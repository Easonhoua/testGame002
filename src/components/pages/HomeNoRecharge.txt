<script setup>
import { ref, onMounted, onUnmounted } from "vue";
import { noRechargeDataRef, noRechargeShowRef } from '@/model/user'

let hours = ref(0);
let minutes = ref(0);
let seconds = ref(0);

const updateCountdown = () => {
    const now = new Date().getTime(); // 当前时间戳
    const targetTime = noRechargeDataRef.value.time; // 目标时间戳

    const diff = targetTime - now; // 计算时间差

    if (diff > 0) {
        hours.value = Math.floor(diff / (1000 * 60 * 60)) >= 10 ? Math.floor(diff / (1000 * 60 * 60)) : `0${Math.floor(diff / (1000 * 60 * 60))}`;
        minutes.value = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60)) >= 10 ? Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60)) : `0${Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60))}`;
        seconds.value = Math.floor((diff % (1000 * 60)) / 1000) >= 10 ? Math.floor((diff % (1000 * 60)) / 1000) : `0${Math.floor((diff % (1000 * 60)) / 1000)}`;
    } else {
        hours.value = '00';
        minutes.value = '00';
        seconds.value = '00';
    }
};

let intervalId;

onMounted(() => {
    updateCountdown();
    intervalId = setInterval(updateCountdown, 1000);
});

onUnmounted(() => {
    clearInterval(intervalId);
});
</script>
